# 顺序队列
## 第一种实现方法
```c
//queue.h
typedef int data_t;
#define N 4

typedef struct{
  data_t data[N];
  int front, rear;
}queue;

queue* queue_create();
int enqueue(queue* q, data_t value);
data_t dequeue(queue* q);
int queue_clear(queue* q);
int queue_empty(queue* q);
int queue_full(queue* q);
queue* queue_free(queue* q);

//queue.c
#include <stdio.h>
#include <stdlib.h>
#include "queue.h"
#include <string.h>

queue* queue_create(){
  queue* q;
  if((q = (queue*)malloc(sizeof(queue))) == NULL){
    printf("malloc failed.\n");
    return NULL;
  }

  q->front = q->rear = 0;
  memset(q->data, 0, sizeof(q->data));

  return q;
}

int enqueue(queue* q, data_t value){
  if(q == NULL){
    printf("q is NULL.\n");
    return -1;
  }

  if((q->rear + 1) % N == q->front){
    printf("The queue is full.\n");
    return -1;
  }

  q->data[q->rear] = value;
  q->rear = (q->rear + 1) % N;

  return 0;
}


data_t dequeue(queue* q){
  if(q == NULL){
    printf("q is NULL.\n");
    return -1;
  }

  if(q->front == q->rear){
    printf("The queue is empty.\n");
    return -1;
  }

  data_t t = q->data[q->front];

  q->front = (q->front + 1) % N;

  return t;
}

int queue_clear(queue* q){
  if(q == NULL){
    printf("q is NULL.\n");
    return -1;
  }

  q->front = q->rear = 0;

  return 0;
}

int queue_empty(queue* q){
  if(q == NULL){
    printf("q is NULL.\n");
    return -1;
  }

  return q->front == q->rear ? 1 : 0;
}

int queue_full(queue* q){
  if(q == NULL){
    printf("q is NULL.\n");
    return -1;
  }

  return (q->rear + 1) % N == q->front ? 1 : 0;
}

queue* queue_free(queue* q){
  if(q == NULL)
    return NULL;

  free(q);
  q = NULL;

  return NULL;
}
```
但是有不足：注意N为4，但实际执行时只能存入3个数据。

## 第二种写法
```c
//queue.h
typedef int data_t;
#define N 4

typedef struct{
  data_t data[N];
  int front, rear;
  /*改进点*/
  int count;
}queue;

queue* queue_create();
int enqueue(queue* q, data_t value);
data_t dequeue(queue* q);
int queue_clear(queue* q);
int queue_empty(queue* q);
int queue_full(queue* q);
queue* queue_free(queue* q);

//queue.c
#include <stdio.h>
#include <stdlib.h>
#include "queue.h"
#include <string.h>

queue* queue_create(){
  queue* q;
  if((q = (queue*)malloc(sizeof(queue))) == NULL){
    printf("malloc failed.\n");
    return NULL;
  }

  q->count = 0;
  q->front = q->rear = 0;
  memset(q->data, 0, sizeof(q->data));

  return q;
}

int enqueue(queue* q, data_t value){
  if(q == NULL){
    printf("q is NULL.\n");
    return -1;
  }

  if(q->count == N){
    printf("The queue is full.\n");
    return -1;
  }

  q->data[q->rear] = value;
  q->rear = (q->rear + 1) % N;
  q->count++;

  return 0;
}


data_t dequeue(queue* q){
  if(q == NULL){
    printf("q is NULL.\n");
    return -1;
  }

  if(q->count == 0){
    printf("The queue is empty.\n");
    return -1;
  }

  data_t t = q->data[q->front];
  q->front = (q->front + 1) % N;
  q->count--;

  return t;
}

int queue_clear(queue* q){
  if(q == NULL){
    printf("q is NULL.\n");
    return -1;
  }

  q->front = q->rear = 0;
  q->count = 0;

  return 0;
}

int queue_empty(queue* q){
  if(q == NULL){
    printf("q is NULL.\n");
    return -1;
  }

  return (q->count == 0) ? 1 : 0;
}

int queue_full(queue* q){
  if(q == NULL){
    printf("q is NULL.\n");
    return -1;
  }

  return (q->count == N) ? 1 : 0;
}

queue* queue_free(queue* q){
  if(q == NULL)
    return NULL;

  free(q);
  q = NULL;

  return NULL;
}
```
此时便可以存入N个数据。
